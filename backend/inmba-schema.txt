/* =========================================================
   CREATE TABLE SQL
========================================================= */

const buildCreateTableSql = (
  tableName,
  schema
) => {

  if (
    !tableName ||
    !schema ||
    !schema.columns
  ) {
    throw new Error(
      `Invalid schema definition for table: ${tableName}`
    );
  }


  const columnSql =
    Object.entries(
      schema.columns
    )
      .map(
        ([
          columnName,
          config,
        ]) => {

          if (
            !/^[A-Za-z0-9_]+$/.test(
              columnName
            )
          ) {
            throw new Error(
              `Unsafe schema column: ${columnName}`
            );
          }

          return (
            `\`${columnName}\` ` +
            config.definition
          );
        }
      )
      .join(",\n");


  return `
    CREATE TABLE IF NOT EXISTS
      \`${tableName}\`
    (
      ${columnSql}
    )
    ENGINE=InnoDB
    DEFAULT CHARSET=utf8mb4
    COLLATE=utf8mb4_unicode_ci
  `;
};
